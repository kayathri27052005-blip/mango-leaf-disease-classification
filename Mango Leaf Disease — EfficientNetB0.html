def load_image(filepath, label):

    image = tf.io.read_file(filepath)

    image = tf.image.decode_image(
        image,
        channels=3,
        expand_animations=False
    )

    image.set_shape([None, None, 3])

    image = tf.image.resize(
        image,
        IMG_SIZE
    )

    image = tf.cast(
        image,
        tf.float32
    )

    return image, label

train_ds = tf.data.Dataset.from_tensor_slices(
    (
        train_df["filepath"].values,
        train_df["label"].values
    )
)

    (
        val_df["filepath"].values,
        val_df["label"].values
    )
)

test_ds = tf.data.Dataset.from_tensor_slices(
    (
        test_df["filepath"].values,
        test_df["label"].values
    )
)

train_ds = train_ds.map(
    load_image,
    num_parallel_calls=tf.data.AUTOTUNE
)

val_ds = val_ds.map(
    load_image,
    num_parallel_calls=tf.data.AUTOTUNE
)

test_ds = test_ds.map(
    load_image,
    num_parallel_calls=tf.data.AUTOTUNE
)

train_ds = train_ds.shuffle(
    1000,
    seed=SEED
)

train_ds = train_ds.batch(BATCH_SIZE).prefetch(
    tf.data.AUTOTUNE
)

val_ds = val_ds.batch(BATCH_SIZE).prefetch(
    tf.data.AUTOTUNE
)

test_ds = test_ds.batch(BATCH_SIZE).prefetch(
    tf.data.AUTOTUNE
)

import os
import numpy as np
import matplotlib.pyplot as plt
from tensorflow.keras.utils import load_img, img_to_array

# ------------------------------------------------------------
# 1. PASTE YOUR FOLDER PATH HERE
# ------------------------------------------------------------

test_folder = r"C:\Users\gayat\Downloads\testing images"


# ------------------------------------------------------------
# 2. FIND ALL IMAGE FILES
# ------------------------------------------------------------

image_extensions = (".jpg", ".jpeg", ".png", ".bmp", ".webp")

image_files = []

for file_name in os.listdir(test_folder):
    if file_name.lower().endswith(image_extensions):
        image_files.append(os.path.join(test_folder, file_name))

image_files = sorted(image_files)

print("Total images found:", len(image_files))


# ------------------------------------------------------------
# 3. PREDICT ALL IMAGES
# ------------------------------------------------------------

results = []

for image_path in image_files:

    # Load image
    img = load_img(image_path, target_size=IMG_SIZE)

    # Convert to array
    img_array = img_to_array(img)

    # Add batch dimension
    img_array = np.expand_dims(img_array, axis=0)

    # Prediction
    predictions = model.predict(img_array, verbose=0)

    predicted_index = np.argmax(predictions[0])
    predicted_class = class_names[predicted_index]
    confidence = predictions[0][predicted_index] * 100

    results.append({
        "image_path": image_path,
        "image_name": os.path.basename(image_path),
        "prediction": predicted_class,
        "confidence": confidence
    })


print("Prediction completed!")
